export async function generateImage(prompt: string, apiKey?: string): Promise<string> {
 const openai = getOpenAIClient(apiKey || undefined);
 const response = await openai.images.generate({
   model: "dall-e-3",
   prompt,
   n: 1,
   size: "1024x1024",
   quality: "standard",
 });
 return response.data[0].url || "";
}

app.post("/api/generate-image", async (req, res) => {
 const { prompt } = req.body;
 const imageUrl = await generateImage(prompt);
 // Fetch and convert image to base64
 const response = await fetch(imageUrl);
 const buffer = await response.arrayBuffer();
 const base64 = Buffer.from(buffer).toString('base64');
  // Store as attachment
 const attachment = await storage.createAttachment({
   cardId: req.body.cardId,
   filename: `generated_${Date.now()}.png`,
   fileType: 'image/png',
   content: base64
 });
  return res.json(attachment);
});
   3. Add frontend handling:


// Detect image requests
const isImageRequest = imageRequestPatterns.some(pattern =>
 pattern.test(userMessage)
);
 
if (isImageRequest) {
 // Generate image
 const response = await fetch('/api/generate-image', {
   method: 'POST',
   body: JSON.stringify({ prompt: userMessage }),
   headers: { 'Content-Type': 'application/json' }
 });
  const attachment = await response.json();
  // Add to chat
 await storage.createChat({
   cardId,
   role: "assistant",
   content: `![Generated image](attachment:${attachment.id})`
 });
}
   4. Display images:


function renderImage(src: string, alt: string) {
 if (src.startsWith('attachment:')) {
   const id = src.substring('attachment:'.length);
   return <img src={`/api/attachments/${id}`} alt={alt} />;
 }
 return <img src={src} alt={alt} />;
}
